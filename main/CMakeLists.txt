# Embed every file under ../web (subdirectories included) as gzip blobs.
# gzip runs at build time; the blobs are pulled into the app with .incbin.
set(WEB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../web")
set(GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/web_gen")
file(GLOB_RECURSE WEB_FILES "${WEB_DIR}/*")
list(SORT WEB_FILES)
if(NOT WEB_FILES)
    message(FATAL_ERROR "web/ has no files to embed")
endif()

file(MAKE_DIRECTORY "${GEN_DIR}")
set(GZ_FILES "")
set(ASM_BODY "")
set(TABLE_DECL "")
set(TABLE_BODY "")
set(IDX 0)
# Re-run this file (and so re-gzip) when any web file changes. New files need a reconfigure.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${WEB_FILES})
foreach(F IN LISTS WEB_FILES)
    file(RELATIVE_PATH REL "${WEB_DIR}" "${F}")
    set(GZ "${GEN_DIR}/${IDX}.gz")
    # gzip at configure time: add_custom_command is not allowed in the component requirement scan.
    execute_process(COMMAND gzip -9 -n -c "${F}"
                    OUTPUT_FILE "${GZ}"
                    RESULT_VARIABLE GZ_RC)
    if(NOT GZ_RC EQUAL 0)
        message(FATAL_ERROR "gzip failed for web/${REL}")
    endif()
    list(APPEND GZ_FILES "${GZ}")
    string(APPEND ASM_BODY
        ".pushsection .rodata\n"
        ".global web_gz_${IDX}_start\n"
        "web_gz_${IDX}_start:\n"
        ".incbin \"${GZ}\"\n"
        ".global web_gz_${IDX}_end\n"
        "web_gz_${IDX}_end:\n"
        ".popsection\n")
    string(APPEND TABLE_DECL
        "extern const uint8_t web_gz_${IDX}_start[];\n"
        "extern const uint8_t web_gz_${IDX}_end[];\n")
    string(APPEND TABLE_BODY
        "    { \"${REL}\", web_gz_${IDX}_start, web_gz_${IDX}_end },\n")
    math(EXPR IDX "${IDX} + 1")
endforeach()
math(EXPR COUNT "${IDX}")

set(ASM_FILE "${GEN_DIR}/web_assets_data.S")
set(C_FILE "${GEN_DIR}/web_assets.c")
file(WRITE "${ASM_FILE}" "${ASM_BODY}")
file(WRITE "${C_FILE}"
    "/* Generated by main/CMakeLists.txt. Do not edit. */\n"
    "#include \"web_assets.h\"\n"
    "${TABLE_DECL}"
    "const web_asset_t web_assets[] = {\n"
    "${TABLE_BODY}"
    "    { NULL, NULL, NULL }\n"
    "};\n"
    "const size_t web_assets_count = ${COUNT};\n")

idf_component_register(
    SRCS "main.c" "config.c" "led.c" "button.c" "lock.c" "usb.c" "usb_fido.c"
         "net.c" "http.c" "recovery.c" "sensing.c" "sensing_dsp.c" "threats.c" "logbuf.c"
         "fido/fido.c" "fido/ctaphid.c" "fido/ctap2.c" "fido/u2f.c" "fido/store.c" "fido/cred.c"
         "${C_FILE}" "${ASM_FILE}"
    INCLUDE_DIRS "." "fido"
    REQUIRES nvs_flash esp_netif esp_wifi bt esp_http_server esp_timer esp_driver_gpio
             led_strip esp_tinyusb cjson lwip cbor mbedtls efuse esp_security esp_driver_tsens)

set_source_files_properties("${ASM_FILE}" PROPERTIES OBJECT_DEPENDS "${GZ_FILES}")
